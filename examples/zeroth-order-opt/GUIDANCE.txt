在 Jetson / Ubuntu 上，最常用的是直接装 build-essential：
```
sudo apt update
sudo apt install build-essential
```
装完后确认：
```
which g++
g++ --version
```
若能看到类似 /usr/bin/g++ 和版本号，再回到 build 目录重新跑：

```
pip install cmake
sudo apt update
sudo apt install build-essential
nvcc --version
conda install -c nvidia cuda-toolkit=13.0
ls /etc/apt/sources.list.d | grep -i cuda || echo "no cuda list"
sudo tee /etc/apt/sources.list.d/cuda-ubuntu2204-x86_64.list >/dev/null <<'EOF'
deb [signed-by=/usr/share/keyrings/cuda-archive-keyring.gpg] https://developer.download.nvidia.com/compute/cuda/repos/ubuntu2204/x86_64/ /
EOF

sudo apt update

apt-cache search cuda-toolkit-13

sudo apt install cuda-toolkit-13-0

apt-cache search nvcc

sudo apt install cuda-nvcc-13-0

dpkg -L cuda-nvcc-13-0 | grep '/nvcc$'

export PATH=/usr/local/cuda-13.0/bin:$PATH
export LD_LIBRARY_PATH=/usr/local/cuda-13.0/lib64:${LD_LIBRARY_PATH:-}

nvcc --version
